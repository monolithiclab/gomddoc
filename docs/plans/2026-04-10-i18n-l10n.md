# Internationalization and Localization Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add i18n/l10n support — translated UI strings via locale bundles, and multi-language content trees with per-language pipeline instances.

**Architecture:** New `internal/locale` package provides a `Bundle` that loads YAML locale files and resolves strings via `T(lang, key)` with fallback. The server scans for BCP 47 directories (e.g., `fr-FR/`) and creates a separate pipeline (provider, metadata, search, navigation) per language. Middleware routes requests by URL prefix to the correct pipeline. Templates use a `T` function for all UI strings. Single-language sites are unaffected.

**Tech Stack:** Go 1.26, `gopkg.in/yaml.v3` (already a dependency), `embed.FS`, `html/template`

**Spec:** `docs/specs/2026-04-10-i18n-l10n.md`

```go
package locale

import (
	"fmt"
	"io/fs"
	"path"
	"strings"

	"gopkg.in/yaml.v3"
)

// Bundle holds locale strings for all loaded languages.
type Bundle struct {
	defaultLang string
	strings     map[string]map[string]string // lang → key → value
}

// T returns the translated string for the given language and key.
// Fallback chain: lang → defaultLang → key itself.
func (b *Bundle) T(lang, key string) string {
	if langStrings, ok := b.strings[lang]; ok {
		if val, ok := langStrings[key]; ok {
			return val
		}
	}
	if langStrings, ok := b.strings[b.defaultLang]; ok {
		if val, ok := langStrings[key]; ok {
			return val
		}
	}
	return key
}

// DefaultLang returns the default language code.
func (b *Bundle) DefaultLang() string {
	return b.defaultLang
}

// Languages returns all loaded language codes.
func (b *Bundle) Languages() []string {
	langs := make([]string, 0, len(b.strings))
	for lang := range b.strings {
		langs = append(langs, lang)
	}
	return langs
}

// LanguageName returns the display name for a language code.
// Falls back to the code itself if language_name is not set.
func (b *Bundle) LanguageName(lang string) string {
	if name := b.T(lang, "language_name"); name != "language_name" {
		return name
	}
	return lang
}

// LoadBundle loads locale YAML files from dir within fsys.
// Files must be named {lang}.yml (e.g., en-US.yml).
func LoadBundle(defaultLang string, fsys fs.FS, dir string) (*Bundle, error) {
	b := &Bundle{
		defaultLang: defaultLang,
		strings:     make(map[string]map[string]string),
	}
	if err := b.loadFrom(fsys, dir); err != nil {
		return nil, err
	}
	return b, nil
}

// MergeFrom loads locale files from another filesystem layer,
// overriding existing keys per language.
func (b *Bundle) MergeFrom(fsys fs.FS, dir string) error {
	return b.loadFrom(fsys, dir)
}

// loadFrom reads all .yml files in dir and merges them into the bundle.
func (b *Bundle) loadFrom(fsys fs.FS, dir string) error {
	entries, err := fs.ReadDir(fsys, dir)
	if err != nil {
		return nil // directory doesn't exist — not an error
	}

	for _, entry := range entries {
		name := entry.Name()
		if entry.IsDir() || !strings.HasSuffix(name, ".yml") {
			continue
		}

		lang := strings.TrimSuffix(name, ".yml")
		filePath := path.Join(dir, name)

		data, err := fs.ReadFile(fsys, filePath)
		if err != nil {
			return fmt.Errorf("read locale %s: %w", filePath, err)
		}

		var values map[string]string
		if err := yaml.Unmarshal(data, &values); err != nil {
			return fmt.Errorf("parse locale %s: %w", filePath, err)
		}

		if b.strings[lang] == nil {
			b.strings[lang] = make(map[string]string)
		}
		for k, v := range values {
			b.strings[lang][k] = v
		}
	}

	return nil
}
```

```go
package locale

import (
	"io/fs"
	"strings"
)

// IsBCP47Dir checks if a directory name is a valid BCP 47 language tag (ll-CC format).
func IsBCP47Dir(name string) bool {
	if len(name) != 5 || name[2] != '-' {
		return false
	}
	// First two chars: lowercase letters
	for _, c := range name[:2] {
		if c < 'a' || c > 'z' {
			return false
		}
	}
	// Last two chars: uppercase letters
	for _, c := range name[3:] {
		if c < 'A' || c > 'Z' {
			return false
		}
	}
	return true
}

// DetectLanguages scans the root of a filesystem for BCP 47 directories.
// Returns the language codes found (e.g., ["fr-FR", "es-ES"]).
// The default language is NOT included (it lives at the root).
func DetectLanguages(fsys fs.FS) []string {
	entries, err := fs.ReadDir(fsys, ".")
	if err != nil {
		return nil
	}

	var langs []string
	for _, entry := range entries {
		if entry.IsDir() && IsBCP47Dir(entry.Name()) {
			langs = append(langs, entry.Name())
		}
	}
	return langs
}

// ExtractLangFromPath extracts a BCP 47 language prefix from a URL path.
// Returns the language code and the remaining path.
// If no language prefix is found, returns ("", originalPath).
func ExtractLangFromPath(urlPath string) (lang, rest string) {
	// Strip leading slash for parsing
	trimmed := strings.TrimPrefix(urlPath, "/")

	var first string
	rest = ""
	if idx := strings.IndexByte(trimmed, '/'); idx >= 0 {
		first = trimmed[:idx]
		rest = trimmed[idx:] // includes leading /
	} else {
		first = trimmed
		rest = "/"
	}

	if IsBCP47Dir(first) {
		return first, rest
	}
	return "", urlPath
}

// ToBCP47 returns the string as-is (already BCP 47 format with dash).
// Provided for clarity at call sites.
func ToBCP47(lang string) string {
	return lang
}
```

```go
// TemplateContext holds the data passed to templates
type TemplateContext struct {
	// Site configuration (public, safe to expose)
	Site *config.SiteConfig

	// Page-specific data (namespaced for extensibility)
	Page PageContext

	// i18n support (set by the server/build when creating the context)
	tFunc     func(string) string // translation function bound to current language
	lang      string              // BCP 47 language code for this request
	languages []LanguageInfo       // all available languages
}

// LanguageInfo holds display information for a language.
type LanguageInfo struct {
	Code   string // BCP 47 code, e.g. "fr-FR"
	Name   string // Display name, e.g. "Français"
	Active bool   // Whether this is the current page's language
}

// T returns the translated string for the given key in the current language.
func (tc *TemplateContext) T(key string) string {
	if tc.tFunc != nil {
		return tc.tFunc(key)
	}
	return key
}

// Lang returns the BCP 47 language code for the current page.
// Page-level frontmatter lang overrides the request-level language.
func (tc *TemplateContext) Lang() string {
	if lang, ok := tc.Page.Meta["lang"].(string); ok && lang != "" {
		return lang
	}
	if tc.lang != "" {
		return tc.lang
	}
	return tc.Site.Language
}

// Languages returns all available languages for the language switcher.
func (tc *TemplateContext) Languages() []LanguageInfo {
	return tc.languages
}

// WithI18n returns a copy of the context with i18n fields set.
// Used by the server and build command when constructing template contexts.
func (tc *TemplateContext) WithI18n(lang string, tFunc func(string) string, languages []LanguageInfo) *TemplateContext {
	tc.lang = lang
	tc.tFunc = tFunc
	tc.languages = languages
	return tc
}
```

```go
// setupLanguagePipelines creates a pipeline per detected language.
// The default language uses the root content; non-default languages use
// their BCP 47 subdirectories (e.g., fr-FR/).
func setupLanguagePipelines(
	cfg *config.Config,
	prov provider.Provider,
	opts PipelineOptions,
	assetsFS fs.FS,
	defaultLang string,
) (*LanguagePipeline, error) {
	// Build default pipeline
	defaultPipeline, err := setupPipeline(cfg, prov, opts)
	if err != nil {
		return nil, fmt.Errorf("default pipeline: %w", err)
	}

	// Load locale bundle: built-in → theme → site overrides
	bundle, err := locale.LoadBundle(defaultLang, assetsFS, "locales")
	if err != nil {
		return nil, fmt.Errorf("load locale bundle: %w", err)
	}

	contentRoot, err := prov.RootFS(context.Background())
	if err != nil {
		return nil, fmt.Errorf("root fs: %w", err)
	}

	// Detect BCP 47 directories
	langs := locale.DetectLanguages(contentRoot)

	lp := &LanguagePipeline{
		Default:   defaultPipeline,
		ByLang:    make(map[string]*Pipeline, len(langs)),
		Bundle:    bundle,
		Languages: langs,
	}

	// Create per-language pipelines
	for _, lang := range langs {
		langFS, err := fs.Sub(contentRoot, lang)
		if err != nil {
			slog.Warn("Failed to create sub-FS for language",
				slog.String("lang", lang), slog.Any("error", err))
			continue
		}

		// Register locale strings for this language (from site .gomddoc/locales/)
		// Theme locales are already loaded via assetsFS

		langProv := provider.NewFSProvider(langFS, cfg.Site.DefaultIndex, cfg.Site.DirIndex, cfg.Site.Exclude)
		langPipeline, err := setupPipeline(cfg, langProv, opts)
		if err != nil {
			slog.Warn("Failed to setup pipeline for language",
				slog.String("lang", lang), slog.Any("error", err))
			continue
		}
		lp.ByLang[lang] = langPipeline

		// Register this language in the bundle if a locale file exists
		// in .gomddoc/locales/ — already handled by MergeFrom in the caller
	}

	// Merge site-level locale overrides
	siteLocalesFS, err := fs.Sub(contentRoot, ".gomddoc")
	if err == nil {
		_ = bundle.MergeFrom(siteLocalesFS, "locales")
	}

	return lp, nil
}
```

```go
package server

import (
	"net/http"
	"strings"
)

// ResolveAPILanguage determines the language for an API request.
// Priority: ?lang= query param > Accept-Language header > default.
// Unknown languages fall back to the default.
func ResolveAPILanguage(r *http.Request, known map[string]bool, defaultLang string) string {
	// 1. Query param
	if lang := r.URL.Query().Get("lang"); lang != "" {
		if known[lang] {
			return lang
		}
		return defaultLang
	}

	// 2. Accept-Language header (take first matching tag)
	if accept := r.Header.Get("Accept-Language"); accept != "" {
		lang := parseAcceptLanguage(accept)
		if lang != "" && known[lang] {
			return lang
		}
	}

	return defaultLang
}

// parseAcceptLanguage extracts the first BCP 47 tag from an Accept-Language header.
// Handles simple cases like "fr-FR", "fr-FR, en-US;q=0.9".
// Does not implement full RFC 4647 matching — just extracts the first tag.
func parseAcceptLanguage(header string) string {
	// Take the first entry (highest priority)
	first, _, _ := strings.Cut(header, ",")
	first = strings.TrimSpace(first)
	// Remove quality value
	tag, _, _ := strings.Cut(first, ";")
	return strings.TrimSpace(tag)
}
```

```go
	indexes     map[string]*search.Index // lang → search index
	defaultLang string

func NewSearchHandler(index *search.Index) *SearchHandler {
	return &SearchHandler{
		indexes:     map[string]*search.Index{"": index},
		defaultLang: "",
	}

// NewMultiLangSearchHandler creates a search handler that dispatches to per-language indexes.
func NewMultiLangSearchHandler(indexes map[string]*search.Index, defaultLang string) *SearchHandler {
	return &SearchHandler{indexes: indexes, defaultLang: defaultLang}
}

func (h *SearchHandler) SearchEndpoint(w http.ResponseWriter, r *http.Request) {
	q := r.URL.Query().Get("q")
	if q == "" {
		writeJSON(w, http.StatusOK, []search.SearchResult{})
		return
	}
	if len(q) > maxQueryLength {
		q = q[:maxQueryLength]
	}

	limit := defaultSearchLimit
	if l := r.URL.Query().Get("limit"); l != "" {
		if parsed, err := strconv.Atoi(l); err == nil && parsed > 0 {
			limit = min(parsed, maxSearchLimit)
		}
	}

	// Resolve language for the search
	known := make(map[string]bool, len(h.indexes))
	for lang := range h.indexes {
		known[lang] = true
	}

	lang := h.defaultLang
	if len(h.indexes) > 1 {
		lang = ResolveAPILanguage(r, known, h.defaultLang)
	}

	idx := h.indexes[lang]
	if idx == nil {
		idx = h.indexes[h.defaultLang]
	}
	if idx == nil {
		writeJSON(w, http.StatusOK, []search.SearchResult{})
		return
	}

	results := idx.Search(q, limit)
	writeJSON(w, http.StatusOK, results)
}
```
